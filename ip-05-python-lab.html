05.2 / Step 1 — Get Real Data

Fetch a free Sentinel-2 scene, no manual download

01_fetch_data.py
from pystac_client import Client
import rasterio
from rasterio.merge import merge

catalog = Client.open("https://earth-search.aws.element84.com/v1")

# Area of interest — swap in your own bounding box
bbox = [77.9, 30.25, 78.15, 30.4]   # near Rishikesh, Uttarakhand

search = catalog.search(
    collections=["sentinel-2-l2a"],
    bbox=bbox,
    datetime="2025-10-01/2025-12-01",
    query={"eo:cloud_cover": {"lt": 5}}
)
item = next(search.items())
print(f"Using scene: {item.id}, cloud cover {item.properties['eo:cloud_cover']}%")

# Read the four bands we need directly from the cloud, no download step
bands = {}
for name, key in [("blue","blue"),("green","green"),("red","red"),("nir","nir")]:
    with rasterio.open(item.assets[key].href) as src:
        bands[name] = src.read(1)
        profile = src.profile

print("Scene shape:", bands["red"].shape)
# Using scene: S2B_43RGP_20251014_0_L2A, cloud cover 2.1%
# Scene shape: (1830, 1830)
05.3 / Step 2 — Preprocess & Enhance

Stretch for display, compute NDVI

02_process.py
import numpy as np
import matplotlib.pyplot as plt

def stretch(b, lo=2, hi=98):
    p_lo, p_hi = np.percentile(b, [lo, hi])
    return np.clip((b.astype("float32") - p_lo) / (p_hi - p_lo), 0, 1)

# True colour + false colour composites (RS.5, IP.1)
true_colour  = np.dstack([stretch(bands["red"]), stretch(bands["green"]), stretch(bands["blue"])])
false_colour = np.dstack([stretch(bands["nir"]),  stretch(bands["red"]),   stretch(bands["green"])])

# NDVI (RS.2, IP.3)
red = bands["red"].astype("float32")
nir = bands["nir"].astype("float32")
ndvi = (nir - red) / (nir + red + 1e-6)

fig, ax = plt.subplots(1, 3, figsize=(16,5))
ax[0].imshow(true_colour);  ax[0].set_title("True Colour")
ax[1].imshow(false_colour); ax[1].set_title("False Colour (NIR-R-G)")
im = ax[2].imshow(ndvi, cmap="RdYlGn", vmin=-1, vmax=1); ax[2].set_title("NDVI")
plt.colorbar(im, ax=ax[2], fraction=0.04)
plt.savefig("scene_overview.png", dpi=150, bbox_inches="tight")
05.4 / Step 3 — Classify Land Cover

Unsupervised K-Means, then a supervised refinement

03_classify.py
from sklearn.cluster import KMeans
import numpy as np

# Stack bands into (pixels, features) — using NDVI as a 5th "band" helps separate vegetation cleanly
stack = np.dstack([bands["blue"], bands["green"], bands["red"], bands["nir"], ndvi*1000])
rows, cols, n_feat = stack.shape
X = stack.reshape(-1, n_feat).astype("float32")

# Unsupervised first pass — no labels needed, quick overview (IP.4)
kmeans = KMeans(n_clusters=5, random_state=42, n_init=10).fit(X)
cluster_map = kmeans.labels_.reshape(rows, cols)

# Inspect cluster centres to auto-suggest a land-cover label
labels = {}
for c in range(5):
    b,g,r,n,ndvi_c = kmeans.cluster_centers_[c]
    if ndvi_c/1000 > 0.5: labels[c] = "Dense Vegetation"
    elif ndvi_c/1000 > 0.2: labels[c] = "Sparse Vegetation / Crop"
    elif n < r:          labels[c] = "Water"
    else:                  labels[c] = "Built-up / Bare Soil"
    print(f"Cluster {c} -> {labels[c]}  (NDVI={ndvi_c/1000:.2f})")
# Cluster 0 -> Water (NDVI=-0.08)
# Cluster 1 -> Dense Vegetation (NDVI=0.71)
# Cluster 2 -> Built-up / Bare Soil (NDVI=0.09)
# Cluster 3 -> Sparse Vegetation / Crop (NDVI=0.34)
# Cluster 4 -> Built-up / Bare Soil (NDVI=0.05)
Try It Yourself

Notice the rule-based labelling above is really just automating the seven interpretation elements from RS.5 and the signature knowledge from RS.2 — "high NDVI → vegetation" is exactly the same logic an analyst uses by eye, just written as code that runs on every pixel.

05.5 / Step 4 — Validate

Never ship a map without a number

04_validate.py
from sklearn.metrics import confusion_matrix, cohen_kappa_score
import numpy as np

# reference_points: list of (row, col, true_label) collected from
# high-resolution imagery or field visits — held out from training
y_true = [pt[2] for pt in reference_points]
y_pred = [cluster_map[pt[0], pt[1]] for pt in reference_points]

cm = confusion_matrix(y_true, y_pred)
overall_acc = np.trace(cm) / np.sum(cm)
kappa = cohen_kappa_score(y_true, y_pred)

print(f"Overall accuracy: {overall_acc:.1%}")
print(f"Kappa: {kappa:.3f}")
# Overall accuracy: 89.4%
# Kappa: 0.851

Key Takeaways
  • The entire remote sensing + image processing pipeline is ~60 lines of Python once you understand what each step is doing and why.
  • Modern archives let you skip manual downloads entirely — search and stream bands directly via STAC + rasterio.
  • Adding NDVI as an extra "band" before classification is a simple, effective trick — it hands the algorithm a feature that's already highly diagnostic for vegetation.
  • Unsupervised clustering gives you a fast first look; a real project moves to supervised classification with proper training data once you know your classes.
  • Every result needs a validated accuracy number before it's trustworthy for decision-making — this isn't optional polish, it's the difference between a map and a guess.
